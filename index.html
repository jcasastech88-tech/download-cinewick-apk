'tv' : 'movie'; const tmdbRes = await fetch(`https://api.themoviedb.org/3/${type}/${id}?api_key=${CONFIG.TMDB_KEY}&language=es-ES`); const tmdbData = await tmdbRes.json(); if(tmdbData.poster_path) imgUrl = `https://image.tmdb.org/t/p/w500${tmdbData.poster_path}`; } catch(e) {} } if (!imgUrl && entry.media$thumbnail) { imgUrl = entry.media$thumbnail.url.replace('/s72-c/', '/w500/'); } if (imgUrl) { const card = document.createElement('div'); card.className = 'poster-card'; card.setAttribute('data-rank', i + 1); // Aquí se pone el número 1, 2, 3... card.innerHTML = ``; track.appendChild(card); } } // Para este diseño de ranking, NO recomiendo duplicar el contenido // porque ver el número "1" dos veces confunde. // Es mejor que el track sea estático o con scroll manual suave. };
